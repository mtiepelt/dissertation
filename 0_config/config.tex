% ---------------------------------------------------
%   ToC Link only numbers
% ---------------------------------------------------
% https://tex.stackexchange.com/questions/574842/change-hyperlinks-in-table-of-contents-to-section-number-and-page-number-only
\makeatletter
\def\numberline#1{%
 \ifx\Hy@tocdestname\ltx@empty
  \hb@xt@\@tempdima{#1\hfil}%
 \else 
 \hb@xt@\@tempdima{\hyper@linkstart{link}{\Hy@tocdestname}\textcolor{black}{#1}\hyper@linkend\hfil}%
 \fi}
\makeatother

}
